- name: Apply custom mods
  run: |
    set -e
    echo "=== Applying mods from /mod ==="

    # [新增] 7. 开启 R8 压缩精简 (7 DEX -> 3 DEX)
    # 将 release 构建类型的 isMinifyEnabled 和 isShrinkResources 改为 true
    GRADLE_FILE="app/build.gradle.kts"
    if [ -f "$GRADLE_FILE" ]; then
      sed -i 's/isMinifyEnabled = false/isMinifyEnabled = true/g' "$GRADLE_FILE"
      sed -i 's/isShrinkResources = false/isShrinkResources = true/g' "$GRADLE_FILE"
      echo "[OK] Enabled R8 Minify and Resource Shrink (for 7->3 DEX)"
    fi

    # 0. 移除 Firebase (保持你原有的逻辑)
    echo "=== Removing Firebase ==="
    sed -i '/id("com\.google\.gms\.google-services")/d' app/build.gradle.kts || true
    sed -i '/libs\.firebase/d' app/build.gradle.kts || true
    sed -i '/firebase/d' gradle/libs.versions.toml || true
    sed -i '/google-services/d' gradle/libs.versions.toml || true
    sed -i '/libs\.plugins\.google/d' build.gradle.kts || true
    sed -i '/libs\.plugins\.firebase/d' build.gradle.kts || true
    find app/src/main/java -type f -name "*.kt" -exec sed -i '/import com\.google\.firebase/d' {} +
    find app/src/main/java -type f -name "*.kt" -exec sed -i 's/, private val analytics: FirebaseAnalytics//' {} +
    find app/src/main/java -type f -name "*.kt" -exec sed -i '/analytics\.logEvent/d' {} +
    find app/src/main/java -type f -name "*.kt" -exec sed -i '/Firebase/d' {} +
    find app/src/main/java -type f -name "*.kt" -exec sed -i '/FirebaseAnalytics/d' {} +
    sed -i '/single.*Firebase/,/}/d' app/src/main/java/me/rerere/rikkahub/di/AppModule.kt || true
    sed -i '/single.*Crashlytics/,/}/d' app/src/main/java/me/rerere/rikkahub/di/AppModule.kt || true
    echo "[OK] Firebase removed"

    # 1. 技能包塞进 assets
    mkdir -p app/src/main/assets/skills
    if [ -d "mod/skills" ]; then
      cp -r mod/skills/* app/src/main/assets/skills/
      echo "[OK] Skills copied to assets/skills/"
    fi

    # 2. 复制 BundledSkillInstaller.java
    if [ -f "mod/BundledSkillInstaller.java" ]; then
      mkdir -p app/src/main/java/me/rerere/rikkahub
      cp mod/BundledSkillInstaller.java app/src/main/java/me/rerere/rikkahub/
      echo "[OK] BundledSkillInstaller.java copied"
    fi

    # 3. 改 RikkaHubApp.kt
    APP_FILE="app/src/main/java/me/rerere/rikkahub/RikkaHubApp.kt"
    if [ -f "$APP_FILE" ]; then
      sed -i '/super\.onCreate()/a \        me.rerere.rikkahub.BundledSkillInstaller.install(this)' "$APP_FILE"
      echo "[OK] Injected install() into RikkaHubApp.kt"
    fi

    # 4. 改 ChatService.kt: 强力绕过 enabledSkills.isEmpty() 检查
    # 策略：直接让 isEmpty() 永远返回 false
    CHAT_FILE="app/src/main/java/me/rerere/rikkahub/service/ChatService.kt"
    if [ -f "$CHAT_FILE" ]; then
       # 匹配 assistant.enabledSkills.isEmpty() 并替换为 false
       sed -i 's/enabledSkills\.isEmpty()/false/g' "$CHAT_FILE"
       echo "[OK] Patched ChatService.kt to bypass isEmpty check"
    fi

    # 5. 改 SkillsToolsKt.kt: 彻底删除过滤逻辑
    # 源码通常是：allSkills.filter { enabledSkills.contains(it.name) }
    # 策略：直接把整个 filter 块替换掉，只保留 allSkills
    SKILL_TOOLS="app/src/main/java/me/rerere/rikkahub/data/ai/tools/SkillsToolsKt.kt"
    if [ -f "$SKILL_TOOLS" ]; then
      # 兼容多种写法，将包含 filter 和 contains 的部分直接替换为 allSkills
      sed -i 's/allSkills\.filter\s*{.*contains.*}/allSkills/g' "$SKILL_TOOLS"
      # 兜底：如果上面没匹配到，尝试更简单的替换
      sed -i 's/enabledSkills\.contains(it\.name)/true/g' "$SKILL_TOOLS"
      echo "[OK] Force-enabled all skills in SkillsToolsKt.kt"
    fi

    # 6. 其他修复 (保持你原有的)
    sed -i '/include(":app:baselineprofile")/d' settings.gradle.kts || true
    echo -e "\n-keep class me.rerere.rikkahub.BundledSkillInstaller { *; }" >> app/proguard-rules.pro
    echo "[OK] Added proguard keep rule"

    echo "=== Mod apply done ==="
